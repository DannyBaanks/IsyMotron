cmake_minimum_required(VERSION 3.22.1)
project(isymotron_gus_jni C)

if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "GUS local inference is currently pinned to arm64-v8a.")
endif()
if(NOT GUS_LLAMA_SOURCE_DIR OR NOT EXISTS "${GUS_LLAMA_SOURCE_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "Set GUS_LLAMA_SOURCE_DIR to llama.cpp commit 842b1880415d6f508f03b789e5ce70194def7bfd.")
endif()
file(READ "${GUS_LLAMA_SOURCE_DIR}/.git/HEAD" _llama_head)
execute_process(COMMAND git -C "${GUS_LLAMA_SOURCE_DIR}" rev-parse HEAD OUTPUT_VARIABLE _llama_revision OUTPUT_STRIP_TRAILING_WHITESPACE)
if(NOT _llama_revision STREQUAL "842b1880415d6f508f03b789e5ce70194def7bfd")
  message(FATAL_ERROR "Wrong llama.cpp revision: ${_llama_revision}")
endif()

set(GGML_NATIVE OFF CACHE BOOL "Disable host-native tuning" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "Keep Android runtime self-contained" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
add_subdirectory("${GUS_LLAMA_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/llama" EXCLUDE_FROM_ALL)

file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/shim/llama")
configure_file("${GUS_LLAMA_SOURCE_DIR}/include/llama.h" "${CMAKE_CURRENT_BINARY_DIR}/shim/llama/llama.h" COPYONLY)
set(GUS_BRIDGE "${CMAKE_CURRENT_LIST_DIR}/../../../../../vendor/gus-runtime/Sources/Model/GUSLlamaBridge.c")
set(GUS_INCLUDE "${CMAKE_CURRENT_LIST_DIR}/../../../../../vendor/gus-runtime/Sources/Model")
add_library(gus_jni SHARED gus_jni.c "${GUS_BRIDGE}")
target_include_directories(gus_jni PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/shim" "${GUS_INCLUDE}" "${GUS_LLAMA_SOURCE_DIR}/include" "${GUS_LLAMA_SOURCE_DIR}/ggml/include")
target_compile_features(gus_jni PRIVATE c_std_11)
target_compile_options(gus_jni PRIVATE -O3 -Wall)
find_library(android-log log)
find_library(android-lib android)
target_link_libraries(gus_jni PRIVATE llama "${android-log}" "${android-lib}")
target_link_options(gus_jni PRIVATE "-Wl,-z,max-page-size=16384")
